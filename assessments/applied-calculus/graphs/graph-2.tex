\documentclass{standalone}

\usepackage{pgfplots}

\usetikzlibrary{angles, quotes, calc}

\pgfplotsset{compat=newest}

\begin{document}
\begin{tikzpicture}
    \begin{axis}[axis lines=center, xlabel=$x$, ylabel=$y$, tick style={draw=none}, xticklabel=\empty, yticklabel=\empty, xmin=0, xmax=2.5, ymin=-1.5, ymax=1.5]
        \pgfmathsetmacro{\h}{2}
        \pgfmathsetmacro{\r}{1}

        \pgfplotsextra{
            \coordinate (A) at (0, 0);
            \coordinate (B) at (\h, \r);
            \coordinate (C) at (\h, 0);

            % triangle
            \draw[thick] (A) -- (B) -- node[right]{$r$} (C);
            \draw[thick] (A) -- (\h, -\r) -- node[right]{$r$} (C);

            % label
            \node[above] at ($(A)!0.6!(C)$) {$h$};

            % angle
            \path pic ["$\theta$", draw, angle radius=1.5cm, angle eccentricity=0.7] {angle=C--A--B};
        }
    \end{axis}
\end{tikzpicture}
\end{document}
