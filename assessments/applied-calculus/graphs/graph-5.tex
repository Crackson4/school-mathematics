\documentclass{standalone}

\usepackage{pgfplots}

\usetikzlibrary{angles, quotes, calc}

\pgfplotsset{compat=newest}

\begin{document}
\begin{tikzpicture}
    \pgfmathsetmacro{\width}{10}
    \pgfmathsetmacro{\height}{10}
    \pgfmathsetmacro{\M}{1.5}
    \pgfmathsetmacro{\m}{2.5}
    \pgfmathsetmacro{\b}{5.5}
    \pgfmathsetmacro{\xi}{4}

    \begin{axis}[axis lines=center, xlabel=$x$, ylabel=$y$, tick style={draw=none}, xticklabel=\empty, yticklabel=\empty, xmin=0, xmax=\width, ymin=-5, ymax=\height]

        % tangent line
        \addplot[thick, domain=0:\width] {\M * x};

        \coordinate (O) at (0, 0);
        \coordinate (A) at (\xi, \M * \xi);
        \coordinate (B) at (\xi, {\m * (\xi - \b)});
        \coordinate (M) at (\xi, 0);

        % axis of rotation
        \addplot[thick, green!80!blue, domain=0:\width] {\m * (x - \b)} node[left, pos=0.9, yshift=3mm] {$L$};

        % ordinals
        \draw[magenta] (A) -- node[left, pos=0.6] {$f(x_i)$} (M) -- node[left] {$m x_i + b$} (B);
        \draw[cyan] (A) -- node[right, yshift=4mm] {$g(x_i)$} ($(\b, 0)!(A)!(0, {-\m * \b})$) coordinate (C);

        % angles
        \path pic[draw, "$\alpha$", angle radius=1cm, angle eccentricity=0.6] {angle=M--O--A};
        \path (\b, 0) coordinate (P) (\b + 1, 0) coordinate (Q) pic[draw, "$\beta$", angle radius=8mm, angle eccentricity=0.6] {angle=Q--P--C};
        \path pic[draw, angle radius=3.5mm] {right angle=O--M--A};
        \path pic[draw, angle radius=3mm] {right angle=P--C--A};

        % labels
        \node[above, xshift=-2mm] at (A) {$A$};
        \node[right, yshift=-1mm] at (B) {$B$};
        \node[right, yshift=-2mm] at (C) {$C$};
    \end{axis}
\end{tikzpicture}
\end{document}
